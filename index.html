async function acceptRequest(requestId, requestData) { if (!currentUser || requestData.toUid !== currentUser.uid) return; try { const id = pairId(requestData.fromUid, requestData.toUid); const reqRef = doc(db, "friendRequests", requestId); const friendshipRef = doc(db, "friendships", id); await runTransaction(db, async (tx) => { const reqSnap = await tx.get(reqRef); const friendshipSnap = await tx.get(friendshipRef); if ( !reqSnap.exists() || reqSnap.data().status !== "pending" || reqSnap.data().toUid !== currentUser.uid ) { throw new Error("طلب الصداقة لم يعد متاحاً."); } if (friendshipSnap.exists()) { throw new Error("علاقة الصداقة موجودة بالفعل."); } tx.set(friendshipRef, { members: [requestData.fromUid, requestData.toUid].sort(), requestId: requestId, createdAt: serverTimestamp() }); tx.update(reqRef, { status: "accepted", respondedAt: serverTimestamp() }); }); status("appStatus", "تم قبول طلب الصداقة بنجاح 💜"); } catch (e) { status("appStatus", err(e), true); } }
